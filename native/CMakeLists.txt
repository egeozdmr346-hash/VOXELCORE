cmake_minimum_required(VERSION 3.18)
project(voxelcore_native VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

option(VOXELCORE_BUILD_JNI "Build the JNI shared library (needs JNI headers)" ON)
option(VOXELCORE_BUILD_TESTS "Build C++ engine tests and benchmark (no JNI needed)" OFF)

# Engine: pure C++, no JNI, unit-testable on its own.
add_library(voxelcore_engine STATIC
    src/engine/cpu.cpp
    src/engine/chunk.cpp
    src/engine/cull.cpp
)
target_include_directories(voxelcore_engine PUBLIC src/engine)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    # No FMA contraction: results must be bit-identical to the Java reference implementations.
    target_compile_options(voxelcore_engine PRIVATE -O2 -ffp-contract=off -Wall -Wextra)
elseif(MSVC)
    target_compile_options(voxelcore_engine PRIVATE /O2 /W4 /fp:precise)
endif()

if(VOXELCORE_BUILD_TESTS)
    enable_testing()
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE voxelcore_engine)
    add_executable(engine_bench tests/engine_bench.cpp)
    target_link_libraries(engine_bench PRIVATE voxelcore_engine)
    add_test(NAME engine_tests COMMAND engine_tests)
endif()

if(NOT VOXELCORE_BUILD_JNI)
    return()
endif()

# JAVA_HOME must point at a JDK (jni.h under $JAVA_HOME/include[/<os>]).
if(DEFINED ENV{JAVA_HOME})
    set(JNI_INCLUDE_DIRS
        "$ENV{JAVA_HOME}/include"
        "$ENV{JAVA_HOME}/include/linux"
        "$ENV{JAVA_HOME}/include/win32"
        "$ENV{JAVA_HOME}/include/darwin"
    )
else()
    find_package(JNI REQUIRED)
endif()

add_library(voxelcore_native SHARED
    src/voxelcore_native.cpp
)
target_link_libraries(voxelcore_native PRIVATE voxelcore_engine)

# com_voxelcore_native_bridge_NativeBridge.h is generated by `javac -h` (Gradle compileJava).
if(NOT JNI_HEADERS_DIR)
    set(JNI_HEADERS_DIR "${CMAKE_SOURCE_DIR}/../build/generated/sources/headers")
endif()

target_include_directories(voxelcore_native PRIVATE
    ${JNI_INCLUDE_DIRS}
    ${JNI_HEADERS_DIR}
    include
)

if(WIN32)
    set_target_properties(voxelcore_native PROPERTIES OUTPUT_NAME "voxelcore_native_windows_x64")
elseif(APPLE)
    set_target_properties(voxelcore_native PROPERTIES OUTPUT_NAME "voxelcore_native_macos")
else()
    set_target_properties(voxelcore_native PROPERTIES OUTPUT_NAME "voxelcore_native_linux_x64")
endif()

set_target_properties(voxelcore_native PROPERTIES PREFIX "")

target_compile_options(voxelcore_native PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -O2>
    $<$<CXX_COMPILER_ID:MSVC>:/W4 /O2>
)
