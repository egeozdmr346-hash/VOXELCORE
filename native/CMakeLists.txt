cmake_minimum_required(VERSION 3.18)
project(voxelcore_native VERSION 0.1.3 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# JAVA_HOME is expected to be set in the environment (it is, on any machine with a JDK
# installed for the Gradle build). We look for jni.h under $JAVA_HOME/include and the
# platform-specific subdirectory (win32/linux/darwin) that ships alongside it.
if(DEFINED ENV{JAVA_HOME})
    set(JNI_INCLUDE_DIRS
        "$ENV{JAVA_HOME}/include"
        "$ENV{JAVA_HOME}/include/linux"
        "$ENV{JAVA_HOME}/include/win32"
        "$ENV{JAVA_HOME}/include/darwin"
    )
else()
    find_package(JNI REQUIRED)
endif()

add_library(voxelcore_native SHARED
    src/voxelcore_native.cpp
)

# The JNI header (com_voxelcore_native_bridge_NativeBridge.h) is machine-generated by
# `javac -h` from NativeBridge.java's native method declarations — it is NOT checked into
# this directory, it lives wherever Gradle's compileJava wrote it. The root build.gradle
# wires this up automatically (Stage 1 TODO: add the CMake-invoking Gradle task); for a
# manual/standalone native build, pass it explicitly:
#   cmake -DJNI_HEADERS_DIR=/path/to/build/native-headers ..
if(NOT JNI_HEADERS_DIR)
    set(JNI_HEADERS_DIR "${CMAKE_SOURCE_DIR}/../build/native-headers")
endif()

target_include_directories(voxelcore_native PRIVATE
    ${JNI_INCLUDE_DIRS}
    ${JNI_HEADERS_DIR}
    include
)

if(WIN32)
    set_target_properties(voxelcore_native PROPERTIES OUTPUT_NAME "voxelcore_native_windows_x64")
elseif(APPLE)
    set_target_properties(voxelcore_native PROPERTIES OUTPUT_NAME "voxelcore_native_macos")
else()
    set_target_properties(voxelcore_native PROPERTIES OUTPUT_NAME "voxelcore_native_linux_x64")
endif()

set_target_properties(voxelcore_native PROPERTIES PREFIX "")

target_compile_options(voxelcore_native PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -O2>
    $<$<CXX_COMPILER_ID:MSVC>:/W4 /O2>
)
